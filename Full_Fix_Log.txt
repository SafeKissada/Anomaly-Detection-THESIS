--------------------------------------------------------------------------------
ปัญหาเดิม
--------------------------------------------------------------------------------
สูตร SSIM ดั้งเดิม (Wang, Bovik, Sheikh, Simoncelli, 2004, IEEE TIP) สมมติว่า
x, y คือค่าความสว่างของภาพที่เป็นบวก อยู่ใน dynamic range คงที่ (เช่น [0,1])
ค่าคงที่ C1=(K1*L)^2, C2=(K2*L)^2 ถูกออกแบบมาให้เสถียรกับ dynamic range นั้น
แต่โค้ดนี้คำนวณ SSIM บน feature ที่ z-score normalize แล้ว (mean~0 ติดลบได้)
ทำให้ C1=0.0001, C2=0.0009 ที่ hardcode ไว้ไม่มีความหมายอ้างอิงกับสเกลจริงของ
ข้อมูลอีกต่อไป

--------------------------------------------------------------------------------
เหตุผลเชิงทฤษฎีที่ใช้แก้ไข
--------------------------------------------------------------------------------
เงื่อนไขที่ต้องคงไว้เพื่อให้สูตร SSIM ใช้ได้ตามทฤษฎี คือ "ข้อมูลนำเข้าต้องอยู่ใน
dynamic range ที่รู้ค่าแน่นอนและเป็นบวก" วิธีแก้ที่เลือกใช้คือ rescale x, y ให้อยู่
ในช่วง [0,1] แบบ dynamic ต่อ batch ก่อนเข้าสูตร โดยคำนวณ min/max ของ x กับ y
"ร่วมกัน" (joint range) ต่อ sample แล้ว affine-transform ทั้งคู่ด้วยค่าเดียวกัน
เพื่อรักษาความสัมพันธ์เชิงสัมพัทธ์ระหว่าง x กับ y ไว้ทุกประการ เมื่อข้อมูลอยู่ใน
[0,1] จริงแล้ว ค่าคงที่ C1=0.01^2, C2=0.03^2 มาตรฐานก็กลับมาใช้ได้ตรงตามทฤษฎี
ต้นฉบับทันที โดยไม่ต้องเดาค่า dynamic range L เอง ค่า min/max ที่ใช้ rescale ถูก
detach() ออกจาก computational graph เพื่อไม่ให้มันกลายเป็นพารามิเตอร์ที่เรียนรู้ได้
โดยไม่ตั้งใจ

  def _ssim_map(self, x, y):
    c = x.shape[1]
    window = self.window.to(device=x.device, dtype=x.dtype).expand(c, 1, self.window_size, self.window_size)
    pad = self.window_size // 2

    mu_x = F.conv2d(x, window, padding=pad, groups=c)
    mu_y = F.conv2d(y, window, padding=pad, groups=c)
    mu_x2, mu_y2, mu_xy = mu_x * mu_x, mu_y * mu_y, mu_x * mu_y

    sigma_x2 = F.conv2d(x * x, window, padding=pad, groups=c) - mu_x2
    sigma_y2 = F.conv2d(y * y, window, padding=pad, groups=c) - mu_y2
    sigma_xy = F.conv2d(x * y, window, padding=pad, groups=c) - mu_xy

    C1, C2 = 0.01 ** 2, 0.03 ** 2
    ssim_map = ((2 * mu_xy + C1) * (2 * sigma_xy + C2)) / \
               ((mu_x2 + mu_y2 + C1) * (sigma_x2 + sigma_y2 + C2) + 1e-12)
    return ssim_map

  def _rescale_to_unit_range(self, x, y):
    # SSIM's C1/C2 constants assume inputs live in a known, bounded, positive
    # dynamic range (classically [0, 1] for image intensities). Here x/y are
    # z-score normalized deep features (mean ~0, unbounded sign), so C1/C2
    # have no meaningful reference scale unless we first map both tensors
    # into a shared [0, 1] range. We compute a joint min/max per-sample
    # (over channel+spatial dims, shared between recon & target so the
    # relative difference between them is preserved) and detach it so it
    # behaves as a fixed rescaling constant rather than a learnable scale.
    with torch.no_grad():
      flat = torch.cat([x, y], dim=1)
      dims = tuple(range(1, flat.dim()))
      lo = flat.amin(dim=dims, keepdim=True)
      hi = flat.amax(dim=dims, keepdim=True)
      data_range = (hi - lo).clamp(min=1e-6)
    x01 = ((x - lo) / data_range).clamp(0.0, 1.0)
    y01 = ((y - lo) / data_range).clamp(0.0, 1.0)
    return x01, y01

  def _ssim_map(self, x, y):
    x, y = self._rescale_to_unit_range(x, y)
    c = x.shape[1]
    window = self.window.to(device=x.device, dtype=x.dtype).expand(c, 1, self.window_size, self.window_size)
    pad = self.window_size // 2

    mu_x = F.conv2d(x, window, padding=pad, groups=c)
    mu_y = F.conv2d(y, window, padding=pad, groups=c)
    mu_x2, mu_y2, mu_xy = mu_x * mu_x, mu_y * mu_y, mu_x * mu_y

    sigma_x2 = F.conv2d(x * x, window, padding=pad, groups=c) - mu_x2
    sigma_y2 = F.conv2d(y * y, window, padding=pad, groups=c) - mu_y2
    sigma_xy = F.conv2d(x * y, window, padding=pad, groups=c) - mu_xy

    # Now that x, y are genuinely in [0, 1], the standard Wang et al. (2004)
    # constants (K1=0.01, K2=0.03, L=1) are valid again.
    C1, C2 = 0.01 ** 2, 0.03 ** 2
    ssim_map = ((2 * mu_xy + C1) * (2 * sigma_xy + C2)) / \
               ((mu_x2 + mu_y2 + C1) * (sigma_x2 + sigma_y2 + C2) + 1e-12)
    return ssim_map

  import re, importlib, sys

  # --- FIX for Findings 2.2 -----------------------------------------------
  # The old per-experiment cells below (previously just bare statements like
  # `TRAIN_DIR : str = "..."`) were DEAD CODE: they only created local
  # variables in this notebook's namespace and had ZERO effect on the actual
  # Config dataclass that scripts/train.py imports and instantiates. Every
  # experiment E0-E8 therefore silently ran with whatever config/config.py
  # happened to contain at checkout time, not the paths shown in the notebook.
  # One of those dead cells (E4) also had a copy-paste bug pointing its
  # SAVE_PATH/OUTPUT_PATH at the E1 result folder instead of E4's.
  #
  # configure_experiment() below REALLY rewrites config/config.py on disk
  # (via regex replace) and then reloads + prints back the values Config()
  # will actually use, so the printed paths are provably what scripts/train.py
  # will read next -- not just notebook-local variables that look right.
  # --------------------------------------------------------------------------

  BASE_DIR   = "/content/drive/MyDrive/Thesis/DATASET/ชุดที่3/Anomaly3"
  RESULT_DIR = "/content/drive/MyDrive/Thesis_Result"
  CONFIG_PATH = "config/config.py"

  def configure_experiment(exp_name, use_grayscale, use_grayscale_eq, loss):
      with open(CONFIG_PATH, encoding="utf-8") as f:
          text = f.read()

      replacements = {
          "TRAIN_DIR":   f"{BASE_DIR}/train_anomaly3",
          "VAL_DIR":     f"{BASE_DIR}/val_anomaly3",
          "TEST_DIR":    f"{BASE_DIR}/test_anomaly3",
          "SAVE_PATH":   f"{RESULT_DIR}/{exp_name}/logs",
          "OUTPUT_PATH": f"{RESULT_DIR}/{exp_name}/visual",
      }
      for field, value in replacements.items():
          text, n = re.subn(
              rf'({field}\s*:\s*str\s*=\s*)([\'\"]).*?\2',
              lambda m: f'{m.group(1)}{m.group(2)}{value}{m.group(2)}',
              text)
          assert n == 1, f"expected exactly 1 match for {field}, got {n}"

      text, n = re.subn(r'(USE_GRAYSCALE\s*:\s*bool\s*=\s*)\w+', rf'\g<1>{use_grayscale}', text)
      assert n == 1
      text, n = re.subn(r'(USE_GRAYSCALE_EQUALIZATION\s*:\s*bool\s*=\s*)\w+', rf'\g<1>{use_grayscale_eq}', text)
      assert n == 1
      text, n = re.subn(r'(LOSS\s*:\s*str\s*=\s*)([\'\"]).*?\2', lambda m: f'{m.group(1)}{m.group(2)}{loss}{m.group(2)}', text)
      assert n == 1

      with open(CONFIG_PATH, "w", encoding="utf-8") as f:
          f.write(text)

      sys.path.insert(0, ".")
      if "config.config" in sys.modules:
          importlib.reload(sys.modules["config.config"])
      from config.config import Config
      cfg = Config()

      print(f"[{exp_name}] TRAIN_DIR   = {cfg.TRAIN_DIR}")
      print(f"[{exp_name}] SAVE_PATH   = {cfg.SAVE_PATH}")
      print(f"[{exp_name}] OUTPUT_PATH = {cfg.OUTPUT_PATH}")
      print(f"[{exp_name}] COLOR_MODE  = {cfg.COLOR_MODE}")
      print(f"[{exp_name}] LOSS        = {cfg.LOSS}")

      assert cfg.SAVE_PATH   == replacements["SAVE_PATH"],   "SAVE_PATH did not update"
      assert cfg.OUTPUT_PATH == replacements["OUTPUT_PATH"], "OUTPUT_PATH did not update"
      return cfg

เหตุผลเชิงทฤษฎี: validation criterion ต้องเป็นตัวประมาณที่ไม่เอนเอียงของ
evaluation criterion (model-selection theory) — resize และ Gaussian blur เป็น
การแปลงแบบไม่เชิงเส้นต่อการจัดอันดับเชิงพื้นที่ ทำให้อันดับภาพจาก raw score กับ
smoothed score ไม่จำเป็นต้องตรงกัน

โค้ดที่แก้: เพิ่ม upsample_and_smooth() เป็นฟังก์ชันกลางใน src/engine.py ให้ทั้ง

ผลทดสอบยืนยัน: คำนวณ score จาก feature/recon คู่เดียวกันผ่านสองเส้นทาง (training

    for the given cfg.AE_MONITOR, using the exact same min/max rule as
    EarlyStopping itself ('val_auroc' -> max, everything else -> min).

    if monitor == 'val_auroc':

2.9 Exception handling อาจ recurse ไม่จำกัด
    ไฟล์ที่แก้: src/data/dataset.py — AnomalyDataset     |     สถานะ: แก้ไขแล้ว

--------------------------------------------------------------------------------
ปัญหาเดิม
--------------------------------------------------------------------------------
ถ้าโหลดภาพ path[idx] ไม่สำเร็จ (ไฟล์เสีย/corrupt) โค้ดจะ log error แล้วสุ่ม
index ใหม่ทั้ง dataset แล้วเรียก __getitem__() ตัวเองซ้ำ "ไม่มีการจำกัดจำนวน
ครั้ง" ถ้าไฟล์เสียจำนวนมาก (เช่น Google Drive หลุดชั่วคราว) มีความเสี่ยง
recursion ลึกผิดปกติ หรือวนซ้ำได้ภาพเดิมซ้ำ ๆ โดยไม่มี log สรุปว่าเกิดขึ้นกี่ครั้ง

--------------------------------------------------------------------------------
เหตุผลที่ใช้แก้ไข
--------------------------------------------------------------------------------
หลัก defensive programming: fallback mechanism ต้องมีขอบเขตชัดเจนเสมอ (bounded
retry) ไม่ใช่ recursion ที่ไม่มีเงื่อนไขหยุด และควรนับสถิติการ fallback ไว้เพื่อ
ตรวจสอบสุขภาพของ data pipeline ได้

--------------------------------------------------------------------------------
โค้ดก่อนแก้ไข
--------------------------------------------------------------------------------

โค้ดหลังแก้ไข
--------------------------------------------------------------------------------
    MAX_LOAD_RETRIES = 5   # (class attribute)

    def _load_one(self, idx):
        """Load+transform a single sample. Raises on failure (no fallback here)."""
        path = self.paths[idx]
        label = self.labels[idx]
        with Image.open(path) as img:
            img = img.convert("RGB")
            ow, oh = img.size
            norm_t = self.norm_tf(img)
            orig_t = self.orig_tf(img)
            preproc_t = self.preproc_tf(img) if self.preproc_tf is not None else orig_t
        return norm_t, orig_t, preproc_t, path, label, (ow, oh)

    def __getitem__(self, idx):
        try:
            return self._load_one(idx)
        except Exception as e:
            logger.error(f"Load failed {self.paths[idx]}: {e}")

        for attempt in range(self.MAX_LOAD_RETRIES):
            random_idx = random.randint(0, len(self) - 1)
            try:
                sample = self._load_one(random_idx)
                self.n_fallbacks += 1
                logger.warning(
                    f"Substituted index {idx} -> {random_idx} "
                    f"(fallback attempt {attempt + 1}/{self.MAX_LOAD_RETRIES}, "
                    f"total fallbacks so far this run: {self.n_fallbacks})")
                return sample
            except Exception as e:
                logger.error(f"Fallback load also failed {self.paths[random_idx]}: {e}")

        raise RuntimeError(
            f"AnomalyDataset: failed to load a usable sample for index {idx} "
            f"after {self.MAX_LOAD_RETRIES} random fallback attempts. "
            f"Too many corrupt/unreachable files in this split — check the "
            f"data source (e.g. a dropped network drive) rather than retrying "
            f"indefinitely.")

  (self.n_fallbacks = 0 ถูกเพิ่มใน __init__() เป็นตัวนับสะสม)

2.10 Grayscale/Equalize อาจทำให้เกิด domain shift (ข้อจำกัดเชิงทฤษฎี)
     ไฟล์ที่แก้: src/data/dataset.py     |     สถานะ: ไม่ใช่บั๊กที่ "แก้ให้หายขาด"
     ได้ด้วยโค้ด — เพิ่ม awareness warning ที่รันทุกครั้งแทน
================================================================================

--------------------------------------------------------------------------------
ทำไมข้อนี้ "แก้ไม่ได้" แบบเดียวกับบั๊กอื่น
--------------------------------------------------------------------------------
ConvNeXtExtractor pretrain มาบนภาพสี ImageNet ปกติ การแปลงภาพเป็น grayscale
หรือ grayscale+equalize ก่อนป้อนเข้า backbone เปลี่ยน texture/intensity
statistics ไปจากที่ backbone เคยเห็นตอน pretrain แม้ extractor.fit_normalization()
จะ refit ค่าเฉลี่ย/ส่วนเบี่ยงเบนมาตรฐานของ "feature" ให้เข้ากับโหมดสีนั้น ๆ
ภายหลัง แต่ "น้ำหนักของ backbone เอง" ไม่ได้ fine-tune ตาม การจะขจัด domain
shift นี้ให้หมดจริง ๆ ต้อง fine-tune backbone (ซึ่งขัดกับการออกแบบที่ freeze
backbone ไว้ทั้งระบบ) จึงเป็นข้อจำกัดเชิงทฤษฎีที่ต้อง "อภิปราย" ไม่ใช่ "แก้"

--------------------------------------------------------------------------------
สิ่งที่แก้ไขได้จริงด้วยโค้ด: เพิ่ม warning log อัตโนมัติ
--------------------------------------------------------------------------------
เพิ่ม log warning ใน build_transforms() ที่ทำงานทุกครั้งที่ COLOR_MODE != 'rgb'
เพื่อเตือนผู้ใช้ (และเป็นหลักฐานในไฟล์ log) ว่ากำลังใช้โหมดที่มีความเสี่ยง domain
shift ให้ต้องตีความผลการทดลองด้วยความระมัดระวัง แทนที่จะปล่อยให้เป็นความรู้ที่
อยู่แต่ในรายงานสถิตที่อาจไม่มีใครเปิดอ่านตอนรันจริง

        # be discussed explicitly in the thesis limitations section when

        logger.warning(

            f"the ImageNet-pretrained backbone sees them. This is a known "
            f"domain-shift risk (see Findings 2.10) — interpret color-mode "
            f"ablation results (e.g. E1/E2/E4/E5/E7/E8) with this in mind.")

  def forward(self, x: torch.Tensor) -> torch.Tensor:
    z = self.encoder(x)
    recon = self.decoder(z)
    if recon.shape != x.shape:
      # The encoder/decoder kernel/stride/padding combination (k4s2p1 x2 +
      # k3s2p1, mirrored on the decoder side) only reproduces the exact
      # input spatial size for feature maps whose H and W are divisible by
      # 8 (true for the default IMAGE_SIZE=224 -> 28x28 ConvNeXt-tiny stage2
      # feature map, but NOT guaranteed for other IMAGE_SIZE values). Rather
      # than letting this surface later as a cryptic broadcasting error
      # inside the loss function, fail immediately here with a clear,
      # actionable message.
      raise RuntimeError(
          f"FeatureAutoencoder: reconstructed shape {tuple(recon.shape)} != "
          f"input shape {tuple(x.shape)}. This almost always means "
          f"cfg.IMAGE_SIZE produces a backbone feature map whose spatial "
          f"size is not evenly divisible by 8 (3 stride-2 downsample "
          f"stages). Pick an IMAGE_SIZE such that "
          f"(IMAGE_SIZE / backbone_downsample_factor) is divisible by 8, "
          f"or reduce the number of downsample stages in "
          f"ConvEncoderWithCNBlock/ConvDecoderWithCNBlock accordingly.")
    return recon

  def bottleneck(self, x: torch.Tensor) -> torch.Tensor:
    return self.encoder(x)

       นี้ "แก้ให้หายขาดด้วยโค้ดไม่ได้" แต่ต้องอภิปรายให้ชัดเจนแทน
